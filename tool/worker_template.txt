import 'package:data_structure_sandbox_browser/stack_sandbox.dart' as stack_api;
import 'package:data_structure_sandbox_browser/queue_sandbox.dart' as queue_api;
import 'generated/@@ID@@/my_linked_list.dart' as list_student;
import 'generated/@@ID@@/list_methods.dart' as list_methods;
import 'generated/@@ID@@/my_bst.dart' as tree_student;
import 'generated/@@ID@@/tree_methods.dart' as tree_methods;
import 'generated/@@ID@@/my_avl.dart' as avl_student;
import 'generated/@@ID@@/avl_methods.dart' as avl_methods;
import 'generated/@@ID@@/my_array_stack.dart' as stack_student;
import 'generated/@@ID@@/stack_methods.dart' as stack_methods;
import 'generated/@@ID@@/my_linked_stack.dart' as linked_stack_student;
import 'generated/@@ID@@/linked_stack_methods.dart' as linked_stack_methods;
import 'generated/@@ID@@/my_linked_queue.dart' as linked_queue_student;
import 'generated/@@ID@@/linked_queue_methods.dart' as linked_queue_methods;
import 'generated/@@ID@@/my_array_queue.dart' as array_queue_student;
import 'generated/@@ID@@/array_queue_methods.dart' as array_queue_methods;

// Batching is bounded for trace/readability; a structure is not capped at 12 nodes.
// A runaway student program is still bounded by the event log, host deadline and
// WebSocket message-size limit, not by its number of nodes.
const maxOperations = 12, maxValue = 999;
const kinds = ['list','tree','avl','stack','linked_stack','linked_queue','array_queue'];
const paths = {
  'list': '@@LIST_PATH@@',
  'tree': '@@TREE_PATH@@',
  'avl': '@@AVL_PATH@@',
  'stack': '@@STACK_PATH@@',
  'linked_stack': '@@LINKED_STACK_PATH@@',
  'linked_queue': '@@LINKED_QUEUE_PATH@@',
  'array_queue': '@@ARRAY_QUEUE_PATH@@',
};

class BrowserSession {
  final String kind;
  final String sessionId = DateTime.now().microsecondsSinceEpoch.toString();
  late Object instance;
  final List<int> reference = [];
  late final List<String> lines;
  late final List<Map<String,dynamic>> methods;

  BrowserSession(this.kind) {
    lines = File(paths[kind]!).readAsLinesSync();
    methods = (jsonDecode(switch(kind) {
      'tree' => tree_methods.discoveredMethodsJson,
      'avl' => avl_methods.discoveredMethodsJson,
      'stack' => stack_methods.discoveredMethodsJson,
      'linked_stack' => linked_stack_methods.discoveredMethodsJson,
      'linked_queue' => linked_queue_methods.discoveredMethodsJson,
      'array_queue' => array_queue_methods.discoveredMethodsJson,
      _ => list_methods.discoveredMethodsJson,
    }) as List).map((entry)=>Map<String,dynamic>.from(entry as Map)).toList();
    reset();
  }

  void reset() {
    reference.clear();
    switch(kind){
      case 'tree': tree_api.TreeRecorder.resetIds(); instance=tree_student.MyBST(); break;
      case 'avl': tree_api.TreeRecorder.resetIds(); instance=avl_student.MyAVL(); break;
      case 'stack': instance=stack_student.MyArrayStack(); break;
      case 'array_queue': instance=array_queue_student.MyArrayQueue(); break;
      case 'linked_stack': list_api.Recorder.resetIds(); instance=linked_stack_student.MyLinkedStack(); break;
      case 'linked_queue': list_api.Recorder.resetIds(); instance=linked_queue_student.MyLinkedQueue(); break;
      default: list_api.Recorder.resetIds(); instance=list_student.MyLinkedList();
    }
  }

  dynamic recorder() => switch(kind) {
    'tree' => tree_api.TreeRecorder(lines)
      ..root = (() => (instance as tree_student.MyBST).root),
    'avl' => tree_api.TreeRecorder(lines)
      ..root = (() => (instance as avl_student.MyAVL).root)
      ..avl = true,
    'stack' => stack_api.StackRecorder(lines)
      ..memory = (() => (instance as stack_student.MyArrayStack).memory)
      ..top = (() => (instance as stack_student.MyArrayStack).top),
    'array_queue' => queue_api.QueueRecorder(lines)
      ..memory = (() => (instance as array_queue_student.MyArrayQueue).memory),
    'linked_stack' => list_api.Recorder(lines)
      ..root = (() => (instance as linked_stack_student.MyLinkedStack).head),
    'linked_queue' => list_api.Recorder(lines)
      ..root = (() => (instance as linked_queue_student.MyLinkedQueue).head)
      ..tail = (() => (instance as linked_queue_student.MyLinkedQueue).tail),
    _ => list_api.Recorder(lines)
      ..root = (() => (instance as list_student.MyLinkedList).head),
  };

  Map<String,Object?> initial() {
    final r=recorder();
    return envelope([r.snapshot()]);
  }

  Map<String,Object?> envelope(List<Map<String,Object?>> steps) => {
    'type':'trace','structure':kind,'sessionId':'$kind:$sessionId',
    'source':{'file':paths[kind], 'lines':lines},
    'steps':steps,'values':[...reference], 'methods':methods,
    'capacity':switch(kind) {
      'stack' => (instance as stack_student.MyArrayStack).memory.length,
      'array_queue' => (instance as array_queue_student.MyArrayQueue).memory.length,
      _ => null,
    },
  };

  Map<String,dynamic> descriptor(String name) => methods.firstWhere(
    (m)=>m['name']==name,orElse:()=>throw FormatException('Unknown method: $name'));

  Object? scalar(Object? raw,String type)=>switch(type) {
    'int' when raw is int && raw.abs()<=maxValue => raw,
    'double' when raw is num && raw.isFinite => raw.toDouble(),
    'bool' when raw is bool => raw,
    'String' when raw is String && raw.length<=256 => raw,
    _ => throw FormatException('Invalid $type argument (integer range: -999…999).'),
  };

  List<Map<String,Object?>> plan(Map<String,dynamic> message) {
    if(message['action']!='run')throw const FormatException('Unknown command.');
    final name=message['method'] as String;
    final params=(descriptor(name)['params'] as List).cast<Map<String,dynamic>>();
    final raw=message['arguments'], legacy=message['values'];
    final List<List<Object?>> calls;
    if(raw is List) {calls=[raw.cast<Object?>()];}
    else if(legacy is List && params.length==1 && params.single['type']=='int'){
      calls=legacy.map((value)=><Object?>[value]).toList();
    }else{throw const FormatException('Supply the method arguments.');}
    if(calls.isEmpty||calls.length>maxOperations)throw const FormatException('Use 1–12 calls.');
    for(final args in calls){
      if(args.length!=params.length)throw const FormatException('Wrong number of arguments.');
    }
    return [for(final args in calls)<String,Object?>{
      'method':name,'args':[for(var i=0;i<args.length;i++)scalar(args[i],params[i]['type'] as String)],
    }];
  }

  Object? invoke(String name,List<Object?> args) => switch(kind){
    'tree' => tree_methods.invokeDiscovered(instance as tree_student.MyBST,name,args),
    'avl' => avl_methods.invokeDiscovered(instance as avl_student.MyAVL,name,args),
    'stack' => stack_methods.invokeDiscovered(instance as stack_student.MyArrayStack,name,args),
    'array_queue' => array_queue_methods.invokeDiscovered(instance as array_queue_student.MyArrayQueue,name,args),
    'linked_stack' => linked_stack_methods.invokeDiscovered(instance as linked_stack_student.MyLinkedStack,name,args),
    'linked_queue' => linked_queue_methods.invokeDiscovered(instance as linked_queue_student.MyLinkedQueue,name,args),
    _ => list_methods.invokeDiscovered(instance as list_student.MyLinkedList,name,args),
  };

  List<int> actualValues() {
    final values=<int>[];
    if(kind=='array_queue') {
      final q=instance as array_queue_student.MyArrayQueue;
      final n=q.memory.length;
      // An invalid student index/size is reported by queueStorageValid();
      // never crash the worker while trying to display its logical values.
      if(n<=0 || q.front<0 || q.front>=n || q.size<0 || q.size>n) return [10000000];
      for(var i=0;i<q.size;i++) {
        final v=q.memory[(q.front+i)%n];
        if(v==null) return [10000000];
        values.add(v);
      }
      return values;
    }
    if(kind=='stack') {
      final s=instance as stack_student.MyArrayStack;
      for(var i=0;i<=s.top && i<s.memory.length;i++) { final v=s.memory[i]; if(v!=null)values.add(v); }
      return values;
    }
    if(kind=='tree'||kind=='avl'){
      final isAvl=kind=='avl';
      // Iterative inorder traversal: a deliberately skewed student BST should
      // not exhaust the worker's Dart call stack or fail a 100-node sentinel.
      final seen=<int>{};
      final pending=<tree_api.TreeNode>[];
      tree_api.TreeNode? current=isAvl
        ?(instance as avl_student.MyAVL).root
        :(instance as tree_student.MyBST).root;
      while(current!=null || pending.isNotEmpty){
        while(current!=null){
          if(!seen.add(current.id)) {
            // A cycle/shared child is an invalid BST, not a reason to hang.
            values.add(10000000);
            return values;
          }
          pending.add(current);
          current=current.left;
        }
        final node=pending.removeLast();
        values.add(node.value);
        current=node.right;
      }
      return values;
    }
    final seen=<int>{};
    var node=switch(kind){
      'linked_stack' => (instance as linked_stack_student.MyLinkedStack).head,
      'linked_queue' => (instance as linked_queue_student.MyLinkedQueue).head,
      _ => (instance as list_student.MyLinkedList).head,
    };
    while(node!=null && seen.add(node.id)){values.add(node.value);node=node.next;}
    if(node!=null)values.add(10000000);
    return values;
  }

  bool updateReference(String name,List<Object?> args,Object? returned) {
    final value=args.length==1 && args[0] is int?args[0] as int:null;
    if(kind=='stack'){
      final stack=reference;
      switch(name){
        case 'push' when value!=null:
          final expected=stack.length<8;
          if(expected)stack.add(value);
          return returned==expected;
        case 'pop':
          final expected=stack.isEmpty?null:stack.removeLast();return returned==expected;
        case 'peek':return returned==(stack.isEmpty?null:stack.last);
        case 'isEmpty':return returned==stack.isEmpty;
        default:stack..clear()..addAll(actualValues());return true;
      }
    }
    if(kind=='array_queue'){
      final q=instance as array_queue_student.MyArrayQueue;
      final values=reference;
      switch(name){
        case 'enqueue' when value!=null:
          final expected=values.length<q.memory.length;
          if(expected) values.add(value);
          return returned==expected;
        case 'dequeue':
          final expected=values.isEmpty?null:values.removeAt(0);
          return returned==expected;
        case 'peek':return returned==(values.isEmpty?null:values.first);
        case 'isEmpty':return returned==values.isEmpty;
        case 'isFull':return returned==(values.length==q.memory.length);
        default:values..clear()..addAll(actualValues());return true;
      }
    }
    if(kind=='linked_stack'){
      switch(name){
        case 'push' when value!=null:
          reference.insert(0,value);return returned==true;
        case 'pop':
          final expected=reference.isEmpty?null:reference.removeAt(0);
          return returned==expected;
        case 'peek':return returned==(reference.isEmpty?null:reference.first);
        case 'isEmpty':return returned==reference.isEmpty;
        default:reference..clear()..addAll(actualValues());return true;
      }
    }
    if(kind=='linked_queue'){
      switch(name){
        case 'enqueue' when value!=null:
          reference.add(value);return returned==true;
        case 'dequeue':
          final expected=reference.isEmpty?null:reference.removeAt(0);
          return returned==expected;
        case 'peek':return returned==(reference.isEmpty?null:reference.first);
        case 'isEmpty':return returned==reference.isEmpty;
        default:reference..clear()..addAll(actualValues());return true;
      }
    }
    if(value==null){reference..clear()..addAll(actualValues());return true;}
    final present=reference.contains(value);
    switch(name){
      case 'insert' when kind=='tree'||kind=='avl':
        if(!present){reference.add(value);reference.sort();}
        return true;
      case 'insert' when kind=='list':
        final index=reference.indexWhere((v)=>v>=value);
        reference.insert(index<0?reference.length:index,value);return true;
      case 'append' when kind=='list':reference.add(value);return true;
      case 'contains':return returned==present;
      case 'remove':
        reference.remove(value);
        return returned==present;
      default:reference..clear()..addAll(actualValues());return true;
    }
  }

  // Physical FIFO invariants: indices are in range; the size identifies
  // exactly the occupied circular slots; rear points to the next free slot.
  // In the full and empty cases front == rear, but size is different.
  bool queueStorageValid(){
    if(kind!='array_queue')return true;
    final q=instance as array_queue_student.MyArrayQueue;
    final m=q.memory, n=m.length;
    if(n<=0 || q.front<0 || q.front>=n || q.rear<0 || q.rear>=n ||
        q.size<0 || q.size>n || q.rear!=(q.front+q.size)%n) return false;
    for(var i=0;i<n;i++){
      final occupied=((i-q.front+n)%n)<q.size;
      if(occupied != (m[i]!=null)) return false;
    }
    return true;
  }

  bool queuePointersValid(){
    if(kind!='linked_queue')return true;
    final q=instance as linked_queue_student.MyLinkedQueue;
    if(q.head==null)return q.tail==null;
    if(q.tail==null)return false;
    final visited=<int>{};
    list_api.ListNode? last, cursor=q.head;
    while(cursor!=null && visited.add(cursor.id)){
      last=cursor;
      cursor=cursor.next;
    }
    // A cycle, an overly long chain, a dangling tail, or tail.next!=null
    // is a physical queue error even when displayed values happen to match.
    return cursor==null && identical(last,q.tail) && q.tail!.next==null;
  }

  Map<String,Object?> execute(Map<String,dynamic> message){
    if(message['action']=='reset'){reset();return initial();}
    final commands=plan(message);
    final r=recorder();
    r.steps.add(r.snapshot());
    for(final command in commands){
      final name=command['method'] as String;
      final args=command['args'] as List<Object?>;
      r.begin(name,args);
      Object? result;
      try {
        switch(kind){
          case 'tree':case 'avl':tree_api.TreeRecorder.active=r as tree_api.TreeRecorder;break;
          case 'stack':stack_api.StackRecorder.active=r as stack_api.StackRecorder;break;
          case 'array_queue':queue_api.QueueRecorder.active=r as queue_api.QueueRecorder;break;
          default:list_api.Recorder.active=r as list_api.Recorder;
        }
        result=invoke(name,args);
      } finally {
        list_api.Recorder.active=null;tree_api.TreeRecorder.active=null;stack_api.StackRecorder.active=null;queue_api.QueueRecorder.active=null;
      }
      final actual=actualValues();
      final standard=(kind=='stack'||kind=='linked_stack') ? {'push','pop','peek','isEmpty'} :
        kind=='array_queue' ? {'enqueue','dequeue','peek','isEmpty','isFull'} :
        kind=='linked_queue' ? {'enqueue','dequeue','peek','isEmpty'} :
        (kind=='tree'||kind=='avl') ? {'insert','contains','remove'} : {'insert','append','contains','remove'};
      final canVerify=standard.contains(name) &&
        (name=='pop'||name=='dequeue'||name=='peek'||name=='isEmpty'||name=='isFull'||(args.length==1&&args.single is int));
      final checked=updateReference(name,args,result);
      // Report structural AVL violations independently of abstract set values.
      // Intermediate rotation snapshots may temporarily be unbalanced: only
      // the state after the public method returns is required to be valid.
      final avlCheck=kind=='avl'
        ? tree_api.auditAvl((instance as avl_student.MyAVL).root) : null;
      final faults=<String>[
        if(avlCheck!=null && avlCheck['acyclic']!=true) 'cycle/shared child',
        if(avlCheck!=null && avlCheck['ordered']!=true) 'BST order',
        if(avlCheck!=null && avlCheck['heights']!=true) 'stored heights',
        if(avlCheck!=null && avlCheck['balanced']!=true) 'balance factor',
      ];
      final ok=checked && actual.join(',')==reference.join(',') &&
        queuePointersValid() && queueStorageValid() && faults.isEmpty;
      r.steps.add(r.snapshot());
      final descriptorReturn=descriptor(name)['returns']=='void';
      final label='$name(${args.join(', ')})';
      final message='$label ${descriptorReturn?'completed':'⇒ $result'} · ${!ok?'CHECK FAILED${faults.isEmpty?'':' (AVL: ${faults.join(', ')})'}':canVerify?'checked':'not automatically verified'}';
      r.end(result,ok,returnedVoid:descriptorReturn,message:message);
      if(!ok)break;
    }
    return envelope(r.steps.cast<Map<String,Object?>>());
  }
}

// This file is generated for exactly one selected student/structure. It is a
// separate Dart process. A stuck method cannot block the browser server.
Future<void> main() async {
  final session = BrowserSession('@@KIND@@');
  stdout.writeln(jsonEncode({'type':'hello','trace':session.initial()}));
  await for (final line in stdin.transform(utf8.decoder).transform(const LineSplitter())) {
    try {
      final message=jsonDecode(line);
      if (message is! Map<String,dynamic>) throw const FormatException('Invalid request');
      if (message['action']=='ping') { stdout.writeln('{"type":"pong"}'); continue; }
      stdout.writeln(jsonEncode(session.execute(message)));
    } catch (error, stack) {
      stderr.writeln('Student execution failed: $error\n$stack');
      stdout.writeln(jsonEncode({'type':'error','message':'$error'}));
    }
  }
}
